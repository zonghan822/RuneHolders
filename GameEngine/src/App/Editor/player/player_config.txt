# Force-Based Physics Properties
# --------------------
# Movement force magnitude applied when pressing WASD (arbitrary units or Newtons)
# Higher = faster acceleration, more responsive
# Lower = slower acceleration, more "heavy" feeling
# Recommended: 600-1200 for responsive feel
movementForce = 800.0

# Friction coefficient for drag/deceleration (0-1 range, dimensionless)
# This creates smooth deceleration when releasing keys
# Higher = stops faster (more friction)
# Lower = more "slippery" movement, longer slide
# Recommended: 0.10-0.20 for smooth feel
frictionCoefficient = 0.15

# Maximum movement speed clamp (pixels per second)
# This is the top speed the player can reach
# Higher = faster top speed
# Lower = slower, more controlled movement
maxSpeed = 300.0

# Player mass (affects acceleration from forces)
# Default is 1.0, increase for "heavier" feel
# Mass affects: acceleration = force / mass
# Higher mass = same force produces less acceleration
mass = 0.8

# Whether player can move initially (true/false)
canMove = true

# Sprite Dimensions

# Dash Mechanics
# --------------------
